[area_match]
# Where to place a listing whose address and name give no area: the nearest
# area position in this file, if within max_km. Sectors are about 2 km
# across; villages and societies on the outskirts spread wider.
# 1.5 km covers a whole sector (centre to corner is ~1.4 km); further than
# that, the nearest listed area was usually wrong in testing (e.g. a
# Diplomatic Enclave daycare landing in Noorpur Shahan).
centres = islamabad_area_centres.csv
max_km = 1.5
# Read CDA sectors in addresses (F-7/4 -> area F-7, sub-area F-7/4)
sectors = cda

[grid]
# Map-grid sweep: each keyword is searched on a map centred on every cell.
# Cells cover the boundary below; cells that don't touch it are skipped.
step_km = 2.5
zoom = 14
# preschool found the most places no other keyword did in the trial
keywords = daycare, preschool

# Islamabad Capital Territory, from OpenStreetMap. Places are kept only if
# their map pin is inside it: this is what excludes Rawalpindi-side areas,
# since Google's addresses often leave out or misstate the city.
