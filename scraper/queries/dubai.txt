# Each keyword is combined with each area: "<keyword> in <area> Dubai".
# Lines starting with # are comments.
#
# Areas are the community names parents search for ("Al Quoz", "Jumeirah"),
# not Dubai Municipality's official spellings ("Al Qouz", "Jumeira"), which
# Google often puts in addresses: those are aliases below. Short forms in
# brackets are aliases too: "Jumeirah Lake Towers (JLT)".
#
# Numbered communities share one area page: "Al Barsha 1" and "Al Barsha
# First" both go to Al Barsha, with "Al Barsha 1" shown on the listing
# ([area_match] numbered = yes). Separate places that only look numbered
# get their own entry (Damac Hills 2).
#
# Sources for where nurseries are: the area filters of edcare.ae and
# edarabia.com nursery listings for Dubai (checked 2026-10-05), plus
# Dubai's residential communities. Industrial zones, free zones with no
# residents and empty desert are left out; the map-grid sweep catches
# nurseries outside these areas.
#
# Not included on purpose:
# - Deira and Bur Dubai as areas: they are districts that span many of the
#   communities below, and an address naming both would be ambiguous.
# - Other emirates' areas with the same names (Al Nahda, Al Khan are also in
#   Sharjah): the Dubai boundary check rejects those.
#
# To check in the trial: whether Google puts the official community after
# the marketing name ("Discovery Gardens - Jabal Ali 1"), which would send
# the listing to the official area (the last name in an address wins).

[boundary]
# Dubai emirate, from OpenStreetMap (includes Hatta, and the sea, which
# doesn't matter for places on land). Places are kept only if their map pin
# is inside it: this is what rejects Sharjah's Al Nahda, Al Khan, Muwaileh
# and the rest, right across the street from Dubai's.
file = dubai_boundary.geojson

[grid]
# Map-grid sweep: the keyword is searched on a map centred on every cell.
# The emirate is mostly desert and sea (1,271 cells), so only cells within
# near_areas_km of an area position below are searched: about 190 cells,
# covering the city and Hatta, and the gaps between the listed areas.
step_km = 2.5
zoom = 14
near_areas_km = 4
# To review after the trial: add a second keyword if "nursery" misses places
keywords = nursery

[area_match]
# Numbered communities roll up: "Al Barsha 1" -> area Al Barsha, sub-area "Al Barsha 1"
numbered = yes
# Where to place a listing whose address and name give no area: the nearest
# area position, if within max_km. Positions from OpenStreetMap
# (python scraper/area_centres.py queries/dubai.txt); four wrong matches
# were corrected by hand, marked "set by hand" in the CSV. Dubai communities
# are 1-3 km across; to tune after the trial.
centres = dubai_area_centres.csv
max_km = 1.5
